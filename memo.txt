angular project set up
sudo npm install -g @angular/cli

create new angular project
ng new projectname --no-strict --standalone false --routing false

1.Which stylesheet format would you like to use it?

2.Do you want to enable Server-Side Rendering(SSR) and Static Site Generation (SSG/Prerendering)?(y/n)
--no

cd foldername

install bootstrap
npm install --save bootstrap@3

ng generate component foldername1/foldername2 --skip-tests
ng g c foldername1/foldername2 --skip-tests
